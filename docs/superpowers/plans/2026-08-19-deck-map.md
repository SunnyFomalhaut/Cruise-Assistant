# 全船甲板手势地图 (Deck Map) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 在应用中实现全船甲板手势地图功能，支持 1~19 层甲板切换、已有图纸（5~9层）的高清矢量缩放平移以及未上线图纸的优雅占位。

**Architecture:** 
- 将 `甲板图/` 下的 SVG 资产迁移至 `app/src/main/assets/decks/` 标准路径；
- 领域层 `DeckMap.kt` 提供甲板信息、图纸可用性判定及资源寻址逻辑；
- UI 层新增 `DeckMapViewModel` 与 `DeckMapScreen`（基于 WebView 硬件加速容器加载 SVG，配合左侧甲板导航 Rail 与复位手势）；
- 在主界面顶栏、设置页及设施指南中提供便捷入口，并在 `CruiseNavHost` 中注册 `Routes.DECK_MAP`。

**Tech Stack:** Kotlin 2.0, Jetpack Compose, Material 3, Android WebView, JUnit 4.

```kotlin
package com.cruise.assistant.domain

/** 甲板信息模型 */
data class DeckInfo(
    val deckNumber: Int,
    val name: String,
    val shortLabel: String,
    val hasMap: Boolean,
    val assetPath: String,
)

/** 全船甲板地图管理仓储 */
object DeckMapRepository {

    /** 目前已拥有 SVG 图纸的甲板集合 */
    val AVAILABLE_DECKS = setOf(5, 6, 7, 8, 9)

    /** 默认最大甲板层数 */
    const val DEFAULT_MAX_DECK = 19

    /**
     * 获取全船甲板列表（从高层到低层降序排列，如 D19 -> D1）
     */
    fun getDecks(maxDeck: Int = DEFAULT_MAX_DECK): List<DeckInfo> {
        val validMax = maxDeck.coerceAtLeast(1)
        return (validMax downTo 1).map { deck ->
            val hasMap = hasDeckMap(deck)
            DeckInfo(
                deckNumber = deck,
                name = "Deck $deck",
                shortLabel = "D$deck",
                hasMap = hasMap,
                assetPath = "decks/deck_$deck.svg",
            )
        }
    }

    /**
     * 判断指定甲板是否具备平面图纸
     */
    fun hasDeckMap(deckNumber: Int): Boolean = deckNumber in AVAILABLE_DECKS

    /**
     * 获取指定甲板的 Android Asset 本地 URL
     */
    fun assetUrlFor(deckNumber: Int): String = "file:///android_asset/decks/deck_$deckNumber.svg"

    /**
     * 获取 HTML 包装内容，以实现响应式缩放与透明背景渲染
     */
    fun htmlWrapperFor(svgUrl: String): String {
        return """
            <!DOCTYPE html>
            <html>
            <head>
                <meta name="viewport" content="width=device-width, initial-scale=1.0, maximum-scale=8.0, user-scalable=yes">
                <style>
                    html, body {
                        margin: 0;
                        padding: 0;
                        width: 100%;
                        height: 100%;
                        background-color: transparent;
                        display: flex;
                        justify-content: center;
                        align-items: flex-start;
                        overflow: auto;
                    }
                    img {
                        width: 100%;
                        max-width: 600px;
                        height: auto;
                        display: block;
                        margin: 16px auto;
                    }
                </style>
            </head>
            <body>
                <img src="$svgUrl" alt="Deck Map" />
            </body>
            </html>
        """.trimIndent()
    }
}
```

package com.cruise.assistant.ui.deckmap

import androidx.lifecycle.ViewModel
import androidx.lifecycle.viewModelScope
import com.cruise.assistant.domain.DeckInfo
import com.cruise.assistant.domain.DeckMapRepository
import kotlinx.coroutines.flow.MutableStateFlow
import kotlinx.coroutines.flow.SharingStarted
import kotlinx.coroutines.flow.StateFlow
import kotlinx.coroutines.flow.combine
import kotlinx.coroutines.flow.stateIn

data class DeckMapUiState(
    val selectedDeck: Int = 5,
    val decks: List<DeckInfo> = emptyList(),
    val hasMap: Boolean = true,
    val assetUrl: String = "",
)

class DeckMapViewModel(
    initialDeck: Int = 5,
    maxDeck: Int = DeckMapRepository.DEFAULT_MAX_DECK,
) : ViewModel() {

    private val _selectedDeck = MutableStateFlow(
        if (DeckMapRepository.hasDeckMap(initialDeck)) initialDeck else 5
    )
    private val _decks = MutableStateFlow(DeckMapRepository.getDecks(maxDeck))

    val state: StateFlow<DeckMapUiState> = combine(
        _selectedDeck,
        _decks,
    ) { currentDeck, deckList ->
        val hasMap = DeckMapRepository.hasDeckMap(currentDeck)
        val url = DeckMapRepository.assetUrlFor(currentDeck)
        DeckMapUiState(
            selectedDeck = currentDeck,
            decks = deckList,
            hasMap = hasMap,
            assetUrl = url,
        )
    }.stateIn(
        scope = viewModelScope,
        started = SharingStarted.WhileSubscribed(5000),
        initialValue = DeckMapUiState(),
    )

    fun onSelectDeck(deckNumber: Int) {
        _selectedDeck.value = deckNumber
    }
}

@Composable
private fun DeckRail(
    decks: List<DeckInfo>,
    selectedDeck: Int,
    onSelectDeck: (Int) -> Unit,
    modifier: Modifier = Modifier,
) {
    val listState = rememberLazyListState()

    // 自动滑动到选中项
    LaunchedEffect(selectedDeck) {
        val index = decks.indexOfFirst { it.deckNumber == selectedDeck }
        if (index >= 0) {
            listState.animateScrollToItem((index - 2).coerceAtLeast(0))
        }
    }

    Surface(
        color = MaterialTheme.colorScheme.surface,
        border = androidx.compose.foundation.BorderStroke(1.dp, Slate200),
        modifier = modifier,
    ) {
        LazyColumn(
            state = listState,
            modifier = Modifier.fillMaxSize(),
            contentPadding = androidx.compose.foundation.layout.PaddingValues(vertical = 8.dp, horizontal = 6.dp),
            verticalArrangement = Arrangement.spacedBy(6.dp),
            horizontalAlignment = Alignment.CenterHorizontally,
        ) {
            items(decks, key = { it.deckNumber }) { deck ->
                val isSelected = deck.deckNumber == selectedDeck
                val containerColor = when {
                    isSelected -> CruiseAzure
                    deck.hasMap -> CruiseAzureContainer.copy(alpha = 0.5f)
                    else -> Slate100
                }
                val textColor = when {
                    isSelected -> Color.White
                    deck.hasMap -> CruiseMarine
                    else -> Slate500
                }

                Box(
                    modifier = Modifier
                        .size(48.dp, 36.dp)
                        .clip(RoundedCornerShape(10.dp))
                        .background(containerColor)
                        .clickable { onSelectDeck(deck.deckNumber) },
                    contentAlignment = Alignment.Center,
                ) {
                    Column(horizontalAlignment = Alignment.CenterHorizontally) {
                        Text(
                            text = deck.shortLabel,
                            style = MaterialTheme.typography.labelMedium,
                            fontWeight = if (isSelected || deck.hasMap) FontWeight.Bold else FontWeight.Normal,
                            color = textColor,
                        )
                        if (deck.hasMap && !isSelected) {
                            Box(
                                modifier = Modifier
                                    .size(4.dp)
                                    .clip(CircleShape)
                                    .background(CruiseAzure),
                            )
                        }
                    }
                }
            }
        }
    }
}

@SuppressLint("SetJavaScriptEnabled")
@Composable
private fun SvgMapView(
    assetUrl: String,
    onWebViewReady: (WebView) -> Unit,
    modifier: Modifier = Modifier,
) {
    AndroidView(
        factory = { context ->
            WebView(context).apply {
                layoutParams = ViewGroup.LayoutParams(
                    ViewGroup.LayoutParams.MATCH_PARENT,
                    ViewGroup.LayoutParams.MATCH_PARENT,
                )
                setBackgroundColor(AndroidColor.TRANSPARENT)
                settings.apply {
                    javaScriptEnabled = false
                    builtInZoomControls = true
                    displayZoomControls = false
                    useWideViewPort = true
                    loadWithOverviewMode = true
                    setSupportZoom(true)
                    allowFileAccess = true
                }
                onWebViewReady(this)
            }
        },
        update = { webView ->
            val html = DeckMapRepository.htmlWrapperFor(assetUrl)
            webView.loadDataWithBaseURL("file:///android_asset/", html, "text/html", "UTF-8", null)
        },
        modifier = modifier,
    )
}
