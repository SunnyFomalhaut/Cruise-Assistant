```kotlin
/**
 * 把 [start, end) 切成"保留段 / 折叠段"交替的序列。
 * 保留段 = 事件占用区间（前后各留 EVENT_PADDING_MINUTES 缓冲）的并集；
 * show 事件按 [rangeOf] 的扩展区间（含 30 分钟等待段）参与。
 * 两个保留段之间超过 MIN_COLLAPSIBLE_GAP_MINUTES 的空档才折叠。
 */
private fun buildCollapsedSpans(
    start: Int,
    end: Int,
    events: List<CruiseEvent>,
    rangeOf: (CruiseEvent) -> Pair<Int, Int>,
): List<Pair<Pair<Int, Int>, Boolean>> {
    val keepRanges = events
        .filter { rangeOf(it).second > start && rangeOf(it).first < end && it.durationMinutes > 0 }
        .map {
            val (eStart, eEnd) = rangeOf(it)
            (eStart - EVENT_PADDING_MINUTES).coerceAtLeast(start) to
                (eEnd + EVENT_PADDING_MINUTES).coerceAtMost(end)
        }
        .sortedBy { it.first }
        .fold(mutableListOf<Pair<Int, Int>>()) { acc, range ->
            val last = acc.lastOrNull()
            if (last != null && range.first <= last.second) {
                acc[acc.lastIndex] = last.first to maxOf(last.second, range.second)
            } else {
                acc += range
            }
            acc
        }

    if (keepRanges.isEmpty()) return listOf((start to end) to true)

    val result = mutableListOf<Pair<Pair<Int, Int>, Boolean>>()
    var cursor = start
    for ((keepStart, keepEnd) in keepRanges) {
        if (keepStart > cursor) {
            val gap = keepStart - cursor
            result += (cursor to keepStart) to (gap >= MIN_COLLAPSIBLE_GAP_MINUTES)
        }
        result += (keepStart to keepEnd) to false
        cursor = keepEnd
    }
    if (cursor < end) {
        val gap = end - cursor
        result += (cursor to end) to (gap >= MIN_COLLAPSIBLE_GAP_MINUTES)
    }

    // 合并相邻的非折叠段，避免碎片
    return result.fold(mutableListOf()) { acc, item ->
        val last = acc.lastOrNull()
        if (last != null && !last.second && !item.second && last.first.second == item.first.first) {
            acc[acc.lastIndex] = (last.first.first to item.first.second) to false
        } else {
            acc += item
        }
    }
}
```

```kotlin
/** 地点下拉的选项：分组标题（餐饮四组）/ 预置条目 / 自定义入口 */
sealed interface LocationPresetOption {
    data class Group(val label: String) : LocationPresetOption
    data class Entry(val entry: LocationCatalogEntry) : LocationPresetOption
    data object Custom : LocationPresetOption
}

/** 按类别 iconKey 展开成下拉选项（meal 按四组分组，组标题不可选） */
fun locationPresetOptions(iconKey: String): List<LocationPresetOption> {
    val entries = BUILT_IN_LOCATION_CATALOG[iconKey] ?: return emptyList()
    if (iconKey != "meal") {
        return entries.map { LocationPresetOption.Entry(it) } + LocationPresetOption.Custom
    }
    return buildList {
        DiningGroup.entries.forEach { group ->
            val groupEntries = entries.filter { it.diningGroup == group }
            if (groupEntries.isNotEmpty()) {
                add(LocationPresetOption.Group(group.label))
                groupEntries.forEach { add(LocationPresetOption.Entry(it)) }
            }
        }
        add(LocationPresetOption.Custom)
    }
}
```

```kotlin
    /**
     * 幂等收敛：是否处于预置地点下拉模式。
     * 非内置船或类别无目录 → false；新建事件（演出/餐饮）→ true（默认下拉）；
     * 编辑事件 → 地点与目录某项匹配、或已在下拉模式时为 true，否则 false（自定义）。
     */
    private fun resolvePresetMode(
        isEditing: Boolean,
        builtInShip: Boolean,
        iconKey: String?,
        currentMode: Boolean,
        deck: Int?,
        zone: ShipZone?,
        placeName: String?,
    ): Boolean {
        if (!builtInShip || iconKey !in BUILT_IN_LOCATION_CATALOG.keys) return false
        if (!isEditing) return true
        val matched = BUILT_IN_LOCATION_CATALOG.getValue(iconKey).any { entry ->
            entry.deck == deck && entry.zone == zone && entry.placeName == placeName
        }
        return matched || currentMode
    }
```
